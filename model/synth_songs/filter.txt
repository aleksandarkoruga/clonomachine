# Quad morph filter: a saw bass (basic table at the saw) through Moog ladder (corner 0,0),
# TB-303 diode (1,0), MS-20 (0,1) and formant (1,1). Env 2 opens the cutoff by 4 octaves per
# note, LFO 1 / 2 move the morph position, resonance high enough to sing.
bpm 140
set volume 0.15
set tableA basic
set a_pos 0.667
set b_on 1
set b_fine 9
set tableB basic
set b_pos 0.667
set filt_on 1
set cutoff 150
set res 7
set slot1 1
set slot1_type 0
set slot1_slope 1
set slot2 2
set slot3 7
set slot3_type 0
set slot4 5
set env2_a 0.002
set env2_d 0.25
set env2_s 0.1
mod env2 cutoff_oct 4
set lfo1_shape 0
set lfo1_rate 0.125
set lfo1_mode 2
set lfo2_shape 1
set lfo2_rate 0.0625
set lfo2_mode 2
mod lfo1 morph_x 1
mod lfo2 morph_y 1
set env1_r 0.05
note 0 33 0.5
note 1 33 0.5
note 1.5 45 0.25
note 2 33 0.75
note 3 36 0.5
note 4 33 0.5
